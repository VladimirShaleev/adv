99
100 /* Concurrency & VM Lifecycle */
101 VM_OPCODE_YIELD = 55, /**< Yields execution so another VM thread can run. */
102 VM_OPCODE_WAIT = 56, /**< Suspends the current execution until its wait condition is met. */
103 VM_OPCODE_HALT = 57, /**< Halts the current VM execution. */
104
105 VM_OPCODE_COUNT_ENUMS = 58, /**< Count of constants in the enumeration. */
106 VM_OPCODE_MAX_ENUM = 0x7FFFFFFF /**< Max value of enum (not used). */
