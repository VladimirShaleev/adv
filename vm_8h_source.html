81/**

94/**
95 * @brief Field information for a reflected structure.
96 * @details Describes the name and type of one structure field. The slot range
97 * is relative to the start of the containing structure.
98 * @ingroup vm_structs
99 */
100typedef struct
101{
102 vm_utf8_t name; /**< UTF-8 field name. */
103 vm_reflect_type_t type; /**< Type of the field value. */
104 vm_uint16_t slot_index; /**< First slot occupied by the field in its containing structure. */
105 vm_uint16_t slot_count; /**< Number of slots occupied by the field. */
106} vm_reflect_field_t;
107
108/**
109 * @brief Reflection information for a structure.
110 * @details The array referenced by @p fields contains @p field_count entries.
111 * Structure values are flattened in field order; nested structures
112 * contribute all of their leaf-field slots. @p slot_count is the
113 * total number of VM slots occupied by the flattened structure.
114 * @ingroup vm_structs
115 */
116struct vm_reflect_struct_t
117{
118 vm_utf8_t name; /**< UTF-8 structure name. */
119 const vm_reflect_field_t* fields; /**< Structure field information. */
120 vm_uint16_t field_count; /**< Number of fields. */
121 vm_uint16_t slot_count; /**< Total number of VM slots occupied by the structure. */
122
123};

125/**
126 * @brief Parameter information for a registered native function.
127 * @details Describes the name and type of one native function parameter. The

134 vm_utf8_t name; /**< UTF-8 parameter name. */

138} vm_reflect_param_t;

140/**
141 * @brief Reflection information for a registered native function.
142 * @details Describes the function name and parameters. @p return_type points
143 * to the single return type when the function returns a value, or is
144 * NULL for a function with no return value. The return value occupies
145 * @p return_slot_count VM slots; zero means there is no return value.
146 * @ingroup vm_structs
147 */
148typedef struct
149{
150 vm_utf8_t name; /**< UTF-8 function name. */
151 const vm_reflect_param_t* params; /**< Function parameter information. */
152 const vm_reflect_type_t* return_type; /**< Single return type, or NULL for no return value. */
153 vm_uint16_t param_count; /**< Number of parameters. */
154 vm_uint16_t return_slot_count; /**< Number of VM slots occupied by the return value. */
155} vm_reflect_func_t;
