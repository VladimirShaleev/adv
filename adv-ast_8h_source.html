18/**
19 * @brief AST node type identifiers.
20 * @details Enumeration of all supported Abstract Syntax Tree node kinds.
21 * @ingroup adv_enums
22 */
23typedef enum
24{
25 ADV_NODE_UNKNOWN = 0, /**< Unknown or uninitialized AST node type. */
26
27 /* Top-level and statements */
28 ADV_NODE_PROGRAM = 1, /**< Root node: stmts.first_stmt -> list of top-level statements. */
29 ADV_NODE_BLOCK = 2, /**< Block `{ ... }`: stmts.first_stmt -> list of block statements. */
30
31 /* Declarations */
32 ADV_NODE_VAR_DECL = 3, /**< `var x: type = expr`: var_decl. */
33 ADV_NODE_NATIVE_DECL = 4, /**< `native name(...)`: func_decl. */
34 ADV_NODE_FUNCTION_DECL = 5, /**< `function name(...)`: func_decl. */
35 ADV_NODE_PARAM_DECL = 6, /**< `a: int`: param_decl (used in func_decl). */
36
37 /* Control flow statements */
38 ADV_NODE_IMPORT_STMT = 7, /**< `import "file"`: import_stmt (path). */
39 ADV_NODE_IF_STMT = 8, /**< `if (cond) then else`: if_stmt. */
40 ADV_NODE_WHILE_STMT = 9, /**< `while (cond) body`: while_stmt. */
41 ADV_NODE_RETURN_STMT = 10, /**< `return expr`: expr_stmt. */
42 ADV_NODE_YIELD_STMT = 11, /**< `yield`: yield statement (no additional union payload). */
43 ADV_NODE_WAIT_STMT = 12, /**< `wait expr`: expr_stmt. */
44 ADV_NODE_EXPR_STMT = 13, /**< Expression statement: expr_stmt. */
45
46 /* Expressions and operations */
47 ADV_NODE_IMPLICIT_CAST_EXPR = 14, /**< Implicit type conversion node inserted during semantic analysis. */
48 ADV_NODE_EXPLICIT_CAST_EXPR = 15, /**< `expr AS type`: explicit cast_expr. */
49 ADV_NODE_IDENTIFIER_EXPR = 16, /**< Identifier reference expression: id_expr. */
50 ADV_NODE_ASSIGN_EXPR = 17, /**< `target = val`: assign_expr. */
51 ADV_NODE_BINARY_EXPR = 18, /**< `a + b`, `a == b`: binary_expr. */
52 ADV_NODE_UNARY_EXPR = 19, /**< `-a`, `!a`: unary_expr. */
53 ADV_NODE_CALL_EXPR = 20, /**< `func(a, b)` or `coroutine[a, b]`: call_expr. */
54 ADV_NODE_MEMBER_ACCESS = 21, /**< `obj.field`: member_access. */
55
56 /* Literals */
57 ADV_NODE_BOOL_LITERAL = 22, /**< Boolean literal: bool_val. */
58 ADV_NODE_INT_LITERAL = 23, /**< 32-bit signed integer literal: int_val. */
59 ADV_NODE_FLOAT_LITERAL = 24, /**< 32-bit floating-point literal: float_val. */
60 ADV_NODE_STRING_LITERAL = 25, /**< String literal handle: str_val. */
61
62 ADV_NODE_COUNT_ENUMS = 26, /**< Count of constants in the enumeration. */
63 ADV_NODE_MAX_ENUM = 0x7FFFFFFF /**< Max value of enum (not used). */
64} adv_node_type_t;

86/**
87 * @brief Abstract Syntax Tree node.
88 * @details Represents a generic node in the AST with type tag, source location, resolved semantic type,
89 * tree hierarchy links, and type-specific union payload.
90 * @ingroup adv_structs
91 */
92struct adv_node_t
93{
94 adv_node_type_t type; /**< Type identifier of the AST node. */

96 vm_value_type_t resolved_type; /**< Type resolved during semantic pass. */
97 adv_node_t* parent; /**< Pointer to parent AST node. */
98 adv_node_t* next_sibling; /**< Pointer to next sibling AST node. */
