18/**
19 * @brief AST node type identifiers.
20 * @details Enumeration of all supported Abstract Syntax Tree node kinds.
21 * @ingroup adv_enums
22 */
23typedef enum
24{
25 ADV_NODE_UNKNOWN = 0, /**< Unknown or uninitialized AST node type. */
26
27 /* Top-level and statements */
28 ADV_NODE_PROGRAM = 1, /**< Root node: stmts.first_stmt -> list of top-level statements. */
29 ADV_NODE_BLOCK = 2, /**< Block `{ ... }`: stmts.first_stmt -> list of block statements. */
30
31 /* Declarations */
32 ADV_NODE_VAR_DECL = 3, /**< `var x: type = expr`: var_decl. */
33 ADV_NODE_NATIVE_DECL = 4, /**< `native name(...)`: func_decl. */
34 ADV_NODE_FUNCTION_DECL = 5, /**< `function name(...)`: func_decl. */
35 ADV_NODE_PARAM_DECL = 6, /**< `a: int`: param_decl (used in func_decl). */
36 ADV_NODE_STRUCT_DECL = 7, /**< `struct Name { ... }`: struct_decl. */
37 ADV_NODE_FIELD_DECL = 8, /**< `name: type`: field_decl. */
38
39 /* Control flow statements */
40 ADV_NODE_IMPORT_STMT = 9, /**< `import "file"`: import_stmt (path). */
41 ADV_NODE_IF_STMT = 10, /**< `if (cond) then else`: if_stmt. */
42 ADV_NODE_WHILE_STMT = 11, /**< `while (cond) body`: while_stmt. */
43 ADV_NODE_RETURN_STMT = 12, /**< `return expr`: expr_stmt. */
44 ADV_NODE_YIELD_STMT = 13, /**< `yield`: yield statement (no additional union payload). */
45 ADV_NODE_WAIT_STMT = 14, /**< `wait expr`: expr_stmt. */
46 ADV_NODE_EXPR_STMT = 15, /**< Expression statement: expr_stmt. */
47
48 /* Expressions and operations */
49 ADV_NODE_IMPLICIT_CAST_EXPR = 16, /**< Implicit type conversion node inserted during semantic analysis. */
50 ADV_NODE_EXPLICIT_CAST_EXPR = 17, /**< `expr AS type`: explicit cast_expr. */
51 ADV_NODE_IDENTIFIER_EXPR = 18, /**< Identifier reference expression: id_expr. */
52 ADV_NODE_ASSIGN_EXPR = 19, /**< `target = val`: assign_expr. */
53 ADV_NODE_BINARY_EXPR = 20, /**< `a + b`, `a == b`: binary_expr. */
54 ADV_NODE_UNARY_EXPR = 21, /**< `-a`, `!a`: unary_expr. */
55 ADV_NODE_CALL_EXPR = 22, /**< `func(a, b)` or `coroutine[a, b]`: call_expr. */
56 ADV_NODE_MEMBER_ACCESS = 23, /**< `obj.field`: member_access. */
57
58 /* Literals */
59 ADV_NODE_BOOL_LITERAL = 24, /**< Boolean literal: bool_val. */
60 ADV_NODE_INT_LITERAL = 25, /**< 32-bit signed integer literal: int_val. */
61 ADV_NODE_FLOAT_LITERAL = 26, /**< 32-bit floating-point literal: float_val. */
62 ADV_NODE_STRING_LITERAL = 27, /**< String literal handle: str_val. */
63
64 ADV_NODE_COUNT_ENUMS = 28, /**< Count of constants in the enumeration. */
65 ADV_NODE_MAX_ENUM = 0x7FFFFFFF /**< Max value of enum (not used). */
66} adv_node_type_t;

88/**
89 * @brief Abstract Syntax Tree node.
90 * @details Represents a generic node in the AST with type tag, source location, resolved semantic type,
91 * tree hierarchy links, and type-specific union payload.
92 * @ingroup adv_structs
93 */
94struct adv_node_t
95{
96 adv_node_type_t type; /**< Type identifier of the AST node. */
97 adv_location_t location; /**< Source location of the node. */
98 adv_symbol_h resolved_type; /**< Handle to the resolved type symbol. */
99 adv_node_t* parent; /**< Pointer to parent AST node. */
100 adv_node_t* next_sibling; /**< Pointer to next sibling AST node. */
101
102 /**
103 * @brief Type-specific payload for the AST node.
104 * @details The active member is determined by the node's type and stores
105 * the data associated with that node kind.
106 */
107 union
108 {
109 /**
110 * @brief Payload for program and block statement lists.
111 **/
112 struct
113 {
114 adv_node_t* first_stmt; /**< Head of statement nodes list. */
115 adv_uint32_t stmt_count; /**< Number of statements in list. */
116 } stmts; /**< Statement list payload. */
117
118 /**
119 * @brief Payload for a variable declaration.
120 **/
121 struct
122 {
123 vm_string_h name; /**< Variable name string handle. */
124 vm_string_h type_hint; /**< Type annotation handle (0 if omitted). */
125 adv_node_t* init_expr; /**< Initializer expression node (nullptr if uninitialized). */
126 adv_bool_t is_global; /**< Flag indicating whether variable is globally scoped. */
127 adv_symbol_h symbol; /**< Handle to associated symbol. */
128 } var_decl; /**< Variable declaration payload. */
129
130 /**
131 * @brief Payload for a function or native declaration.
132 **/
133 struct
134 {
135 vm_string_h name; /**< Function name string handle. */
136 vm_string_h return_type_hint; /**< Return type annotation handle. */
137 adv_node_t* first_param; /**< Head of parameter declarations list (linked by next_sibling). */
138 adv_uint32_t param_count; /**< Number of parameters. */
139 adv_node_t* body; /**< Pointer to BLOCK node. */
140 adv_symbol_h symbol; /**< Handle to associated symbol. */
141 } func_decl; /**< Function declaration payload. */
142
143 /**
144 * @brief Payload for a structure declaration.
145 **/
146 struct
147 {
148 vm_string_h name; /**< Structure name string handle. */
149 adv_node_t* first_field; /**< Head of field declarations list. */
150 adv_uint32_t field_count; /**< Number of field declarations. */
151 adv_symbol_h symbol; /**< Handle to associated symbol. */
152 } struct_decl; /**< Structure declaration payload. */
153
154 /**
155 * @brief Payload for a structure field declaration.
156 **/
157 struct
158 {
159 vm_string_h name; /**< Field name string handle. */
160 vm_string_h type_hint; /**< Field type annotation handle. */
161 adv_symbol_h symbol; /**< Handle to associated symbol. */
162 } field_decl; /**< Structure field declaration payload. */
163
164 /**
165 * @brief Payload for a function parameter declaration.
166 **/
167 struct
168 {
169 vm_string_h name; /**< Parameter name string handle. */
170 vm_string_h type_hint; /**< Type annotation handle (0 if omitted). */
171 adv_symbol_h symbol; /**< Handle to associated symbol. */
172 } param_decl; /**< Function parameter declaration payload. */
173
174 /**
175 * @brief Payload for an import statement.
176 **/
177 struct
178 {
179 adv_node_t* root; /**< Root node of imported AST. */
180 vm_string_h path; /**< Imported module or file path string handle. */
181 } import_stmt; /**< Import statement payload. */
182
183 /**
184 * @brief Payload for a conditional statement.
185 **/
186 struct
187 {
188 adv_node_t* cond; /**< Condition expression. */
189 adv_node_t* then_branch; /**< Execution block if condition evaluates to true. */
190 adv_node_t* else_branch; /**< Optional else block or nested ADV_NODE_IF_STMT. */
191 } if_stmt; /**< Conditional statement payload. */
192
193 /**
194 * @brief Payload for a while loop statement.
195 **/
196 struct
197 {
198 adv_node_t* cond; /**< Loop condition expression. */
199 adv_node_t* body; /**< Pointer to BLOCK node executed per iteration. */
200 } while_stmt; /**< Loop statement payload. */
201
202 /**
203 * @brief Payload for expression, return, and wait statements.
204 **/
205 struct
206 {
207 adv_node_t* expr; /**< Expression payload (nullptr if omitted/void). */
208 } expr_stmt; /**< Expression statement payload. */
209
210 /**
211 * @brief Payload for implicit and explicit cast expressions.
212 **/
213 struct
214 {
215 adv_node_t* operand; /**< Expression being converted. */
216 vm_string_h target_type_hint; /**< Explicit target type string handle. */
217 adv_symbol_h from_type; /**< Original value type. */
218 adv_symbol_h to_type; /**< Target converted type. */
219 } cast_expr; /**< Type cast expression payload. */
220
221 /**
222 * @brief Payload for an identifier reference expression.
223 **/
224 struct
225 {
226 vm_string_h name; /**< Identifier name string handle. */
227 vm_string_h type_hint; /**< Type annotation handle (0 if omitted). */
228 adv_symbol_h symbol; /**< Handle to associated symbol. */
229 } id_expr; /**< Identifier reference expression payload. */
230
231 /**
232 * @brief Payload for an assignment expression.
233 **/
234 struct
235 {
236 adv_node_t* target; /**< Target node (IDENTIFIER_EXPR or MEMBER_ACCESS). */
237 adv_node_t* val; /**< Value expression to assign. */
238 } assign_expr; /**< Assignment expression payload. */
239
240 /**
241 * @brief Payload for a binary operator expression.
242 **/
243 struct
244 {
245 adv_uint32_t op; /**< Operator token code (e.g. '+', EQ). */
246 adv_node_t* left; /**< Left-hand operand. */
247 adv_node_t* right; /**< Right-hand operand. */
248 } binary_expr; /**< Binary operator expression payload. */
249
250 /**
251 * @brief Payload for a unary operator expression.
252 **/
253 struct
254 {
255 adv_uint32_t op; /**< Operator token code. */
256 adv_node_t* operand; /**< Target operand. */
257 } unary_expr; /**< Unary operator expression payload. */
258
259 /**
260 * @brief Payload for a function or coroutine call expression.
261 **/
262 struct
263 {
264 vm_string_h name; /**< Function or coroutine name string handle. */
265 adv_node_t* first_arg; /**< Head of argument expressions list (linked by next_sibling). */
266 adv_uint32_t arg_count; /**< Number of arguments passed. */
267 adv_bool_t is_async; /**< True for asynchronous invocation `coroutine[a, b]`, false for standard `func(a, b)`. */
268 adv_symbol_h symbol; /**< Handle to associated symbol. */
269 } call_expr; /**< Call expression payload. */
270
271 /**
272 * @brief Payload for a member access expression.
273 **/
274 struct
275 {
276 adv_node_t* target; /**< Target object instance expression. */
277 vm_string_h member; /**< Member field name string handle. */
278 adv_symbol_h symbol; /**< Handle to the resolved field symbol. */
279 } member_access; /**< Member access expression payload. */
280
281 adv_bool_t bool_val; /**< Boolean literal value. */
282 adv_sint32_t int_val; /**< Integer literal value. */
283 adv_float32_t float_val; /**< Floating-point literal value. */
284 vm_string_h str_val; /**< String literal handle. */
285 }
286#ifdef DOXYGEN
287 unnamed; /**< Node payload. */
288#endif
