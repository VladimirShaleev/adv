56/**
57 * @brief Encodes version components into a single integer.
58 * @details Combines major, minor, and micro versions into a 32-bit value:
59 * - Bits 24-31: Major version (0-255)
60 * - Bits 16-23: Minor version (0-255)
61 * - Bits 0-15: Micro version (0-65535)
62 * The encoded value can be compared numerically when each component
63 * is within its documented range.
64 * @param[in] major Major version number.
65 * @param[in] minor Minor version number.
66 * @param[in] micro Micro version number.
67 * @return Encoded version as unsigned long.
68 * @sa ADV_VERSION
69 * @ingroup adv_macros
70 */
71#define ADV_VERSION_ENCODE(major, minor, micro)
72 ((((unsigned long)(major)) << 24) | (((unsigned long)(minor)) << 16) | (unsigned long)(micro))
73
74/**
75 * @brief Internal macro for string version generation.
76 * @details Helper macro that stringizes version components (e.g., 1, 0, 0 -> "1.0.0").
77 * @param[in] major Major version number.
78 * @param[in] minor Minor version number.

81 * @note For internal use only.
