32/**

52/**
53 * @brief Callback that supplies source files to the compiler.
54 * @details Called first when the compiler resolves an import. The callback
55 * may obtain the source in any way, including allocating it in memory
56 * or reading it from a file.
57 * @param[in] name Import or source name requested by the compiler.
58 * @param[in] depth Import nesting level; the primary source is at level zero.
59 * @param[in] data User data specified when setting the callback.
60 * @return Pointer to the requested source, or `NULL` to let the compiler
61 * continue searching the provided sources and file-system paths.
62 * @note The compiler reads and parses a returned source before calling
63 * ::adv_release_import_callback_t, if a release callback is set.
64 * The client does not need to keep the source data alive for the
65 * remainder of the compilation.
66 * @sa ::adv_release_import_callback_t
67 * @ingroup adv_callbacks
68 */
69typedef adv_source_t*
70(*adv_import_callback_t)(adv_utf8_t name,
71 adv_uint32_t depth,
72 adv_data_t data);
