4 * @details Defines the binary script header and section table, along with

23/**
24 * @brief Header of a compiled VM script.
25 * @details The magic field is stored as four raw bytes. The version and
26 * checksum fields are always stored in little-endian byte order.
27 * Big-endian hosts must use the byte-swapping helpers when reading
28 * or writing multibyte values.
29 * @ingroup vm_structs
30 */
31typedef struct
32{
33 vm_char_t magic[4]; /**< Four-byte script format identifier. */
34 vm_uint32_t version; /**< Script format version, stored little-endian. */
35 vm_uint32_t checksum; /**< Adler-32 checksum, stored little-endian. */
36} VM_PACKED_ATTR vm_script_header_t;

38/**
39 * @brief Offsets and sizes of the script data and bytecode sections.
40 * @details The data block contains string entries. Each entry starts with a
41 * little-endian vm_uint16_t string length, excluding the terminating
42 * null character, followed by the string and its null terminator.
43 * The text block contains bytecode made up of vm_opcode_t opcodes
44 * and their operands. All multibyte values, including these fields
45 * and operands, are stored in little-endian byte order. Big-endian
46 * hosts must convert multibyte values with the byte-swapping helpers.
47 * @ingroup vm_structs
48 */
49typedef struct
50{
51 vm_uint32_t export_offset; /**< Byte offfset of the export symbols block. */
52 vm_uint32_t export_size; /**< Size in bytes of the export symbols block */
53 vm_uint32_t data_offset; /**< Byte offset of the string data block. */
54 vm_uint32_t data_size; /**< Size in bytes of the string data block. */
55 vm_uint32_t text_offset; /**< Byte offset of the bytecode block. */
56 vm_uint32_t text_size; /**< Size in bytes of the bytecode block. */
57} VM_PACKED_ATTR vm_script_table_t;

59/**
60 * @brief Export table metadata for the script symbol block.
61 * @details Stores the number and byte size of exported variables and
62 * functions. Values are kept in little-endian byte order and are

64 * @ingroup vm_structs

66typedef struct
67{
68 vm_uint32_t module_name; /**< Offset of the module name string in the data section. */

70 vm_uint16_t var_count; /**< Number of exported variables. */
71 vm_uint16_t var_size; /**< Size in bytes of the variable export block. */
72 vm_uint16_t func_count; /**< Number of exported functions. */
73 vm_uint16_t func_size; /**< Size in bytes of the function export block. */
74 vm_uint16_t native_count; /**< Number of exported native functions. */
75 vm_uint16_t native_size; /**< Size in bytes of the native function export block. */
76} VM_PACKED_ATTR vm_script_export_t;

139/**
140 * @name Functions for Script Handling.
141 * @brief Functions for handling compiled VM scripts.
142 * @note Serialized script data is always stored in little-endian byte order
143 * in the binary format. The `vm_read_*` helpers read the raw payload,
144 * convert it to the native byte order of the current host, and return
145 * values in host-endian representation. In other words, callers always
146 * receive `vm_uint16_t`, `vm_uint32_t`, `vm_sint32_t`, and floating-point
147 * values as native CPU-order values, while the on-disk format remains
148 * little-endian for all `int16`, `int32`, and larger fields.
149 * @{
150 */
