80/**
81 * @brief Compiles a source file or in-memory source.
82 * @details If @p file is not `NULL` and is not empty, it identifies the
83 * primary source file. Otherwise, the compiler takes the primary
84 * source from @p sources. If neither is provided, there is no
85 * source to compile.
86 *
87 * Imports are resolved first through the importer configured with
88 * ::adv_options_set_importer, then through the entries in
89 * @p sources. If neither supplies the module, the compiler searches
90 * the file system, including the importing file's directory, the
91 * configured import directories, and the current working directory.
92 *
93 * The returned ::adv_result_t reports whether compilation succeeded
94 * regardless of whether @p result is `NULL`. When @p result is not
95 * `NULL`, it receives a compilation result object containing
96 * diagnostic messages and, when available, the syntax tree and
97 * symbol table.
98 * @param[in] compiler Compiler instance.
99 * @param[in] file Path to the primary `.adv` file, or `NULL`/empty to compile
100 * the primary source from @p sources.
101 * @param[in] source_count Number of entries in @p sources.
102 * @param[in] sources In-memory source files. May be `NULL` when
103 * @p source_count is zero.
104 * @param[in] options Compilation options, or `NULL` to use default options.
105 * @param[out] result Receives the compilation result object, or may be `NULL`
106 * if detailed diagnostics and compilation data are not needed.
107 * @return ::ADV_RESULT_SUCCESS on success, or an error result describing
108 * why compilation did not succeed.
109 * @note To compile entirely from memory, pass an empty @p file and provide
110 * the primary file in @p sources.
111 * @ingroup adv_functions
112 */
113ADV_API adv_result_t
114adv_compiler_compile(adv_compiler_t compiler,
115 adv_utf8_t file,
116 adv_uint32_t source_count,
117 const adv_source_t* sources,
118 adv_options_t options,
119 adv_compilation_result_t* result);
